## 1. Objective & Overview

The objective of Assignment 1 is to construct a client-side **Notes / Todo Application** utilizing pure HTML5, CSS3, and JavaScript, demonstrating proficiency with:
1. **HTML5 Web Storage API (`localStorage`)**: Persisting structured data across browser sessions and tabs with zero database server requirements.
2. **JSON Serialization & Deserialization (`JSON.stringify()` & `JSON.parse()`)**: Translating complex in-memory JavaScript object hierarchies into string-based key-value pairs and reconstructing them upon page initialization.
3. **Full CRUD Lifecycle**: Creating, Reading, Updating, and Deleting persistent records on the client side.

### A. Note Object Representation

```javascript
  createdAt: "2026-11-26T04:20:00.000Z",
  updatedAt: null
};
```

### B. Storage Helper Functions
```javascript
const STORAGE_KEY = "notes_app_data";

// Read from LocalStorage
function getNotes() {
  const raw = localStorage.getItem(STORAGE_KEY);
  return raw ? JSON.parse(raw) : [];
}

// Write to LocalStorage
function saveNotes(notes) {
  localStorage.setItem(STORAGE_KEY, JSON.stringify(notes));
}
```

### C. JSON Backup & File Import Engine
```javascript
// Export JSON file
function exportNotes() {
  const notes = getNotes();
  const blob = new Blob([JSON.stringify(notes, null, 2)], { type: 'application/json' });
  const url = URL.createObjectURL(blob);
  const a = document.createElement('a');
  a.href = url;
  a.download = `notes_backup_${Date.now()}.json`;
  a.click();
  URL.revokeObjectURL(url);
}

// Import JSON file
function importNotes(file) {
  const reader = new FileReader();
  reader.onload = (e) => {
    const data = JSON.parse(e.target.result);
    if (Array.isArray(data)) {
      saveNotes(data);
      renderNotes();
    }
  };
  reader.readAsText(file);
}
```
