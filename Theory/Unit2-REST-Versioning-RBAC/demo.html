function authorize(...allowedRoles) {
  return (req, res, next) => {
    if (!allowedRoles.includes(req.user.role)) {
      return res.status(403).json({
        success: false,
        error: `Forbidden: role '${req.user.role}' cannot access this endpoint.`
      });
    }
    next();
  };
}

app.get('/api/v1/users', authenticate, authorize('student', 'faculty', 'admin'), (req, res) => {
  res.json(records.map(({ id, name, role }) => ({ id, name, role })));
});

app.get('/api/v2/users', authenticate, authorize('student', 'faculty', 'admin'), (req, res) => {
  res.json({
    success: true,
    apiVersion: 'v2',
    meta: { count: records.length, timestamp: new Date().toISOString() },
    data: records
  });
});
