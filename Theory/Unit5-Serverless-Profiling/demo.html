exports.handler = async (event, context = {}) => {
  const start = performance.now();
  const coldStart = !isWarm;
  isWarm = true;
  if (coldStart) await new Promise((r) => setTimeout(r, 120));
  return {
    statusCode: 200,
    body: JSON.stringify({
      coldStart,
      durationMs: Number((performance.now() - start).toFixed(3))
    })
  };
};
