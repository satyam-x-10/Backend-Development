DROP TABLE IF EXISTS products CASCADE;

CREATE TABLE products (
    id INTEGER GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    name VARCHAR(100) NOT NULL,
    category VARCHAR(50) NOT NULL,
    price NUMERIC(10,2) NOT NULL,
    attributes JSONB
);

-- Category-specific attribute filtering
SELECT name, price, attributes ->> 'author' AS author, (attributes ->> 'pages')::INT AS pages
FROM products WHERE category = 'book' AND (attributes ->> 'pages')::INT > 500;

SELECT name, price, attributes ->> 'cpu' AS processor, (attributes ->> 'ram_gb')::INT AS ram
FROM products WHERE category = 'laptop' AND (attributes ->> 'ram_gb')::INT >= 32;

SELECT name, price, attributes ->> 'color' AS color, (attributes ->> 'dpi')::INT AS dpi
FROM products WHERE category = 'accessory' AND (attributes ->> 'dpi')::INT >= 4000;

-- Containment query (@>)
SELECT id, name, category, price, attributes
FROM products WHERE attributes @> '{"wireless": true}';

-- Dynamic JSONB update without ALTER TABLE
UPDATE products
SET attributes = attributes || '{"discount_pct": 15, "in_stock": true}'::JSONB
WHERE name = 'Logitech MX Master 3S';

-- GIN Index and Execution Plan comparison
EXPLAIN ANALYZE SELECT name FROM products WHERE attributes @> '{"wireless": true}';
CREATE INDEX idx_products_attributes ON products USING GIN (attributes);
