Dockerfile and GitHub Actions Pipeline

# Dockerfile
FROM node:20-alpine
WORKDIR /app
COPY package*.json ./
RUN npm install --omit=dev
COPY . .
EXPOSE 4005
HEALTHCHECK --interval=30s --timeout=5s CMD wget -qO- http://localhost:4005/health || exit 1
CMD ["node", "server.js"]
# ci-workflow.yml
jobs:
  test-and-build:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-node@v4
        with: { node-version: '20' }
      - run: npm install
      - run: npm test
      - run: docker build -t unit4-backend:ci .
