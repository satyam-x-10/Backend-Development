eventBus.on('order.created', (order) => {
  const entry = `[Async Event] Notification sent to ${order.student} for ${order.item}`;

app.post('/orders', (req, res) => {
  const userService = serviceRegistry.get('user-service');
  if (!userService) {
    return res.status(503).json({ error: 'Service Discovery failed: user-service unavailable' });
  }
  const order = { orderId: 1042, student: req.body.student, verifiedVia: userService.url };
  eventBus.emit('order.created', order);
  res.status(201).json({ order });
});
